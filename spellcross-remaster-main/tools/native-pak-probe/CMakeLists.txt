cmake_minimum_required(VERSION 3.20)
project(SpellcrossNativePakProbe LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
find_package(wxWidgets 3.2 REQUIRED COMPONENTS core base)
include(${wxWidgets_USE_FILE})
add_executable(native_pak_probe main.cpp)
target_link_libraries(native_pak_probe PRIVATE ${wxWidgets_LIBRARIES})
include(CTest)
if(BUILD_TESTING)
  add_executable(native_pak_tests tests.cpp)
  target_link_libraries(native_pak_tests PRIVATE ${wxWidgets_LIBRARIES})
  add_test(NAME native_pak COMMAND native_pak_tests)
endif()
